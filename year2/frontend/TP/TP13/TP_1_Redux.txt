7 & 8. panierReducer.js

const initialState = { items: [] }; [cite: 20]

const panierReducer = (state = initialState, action) => {
  switch (action.type) {
    case 'ADDLIVRE':
      return { ...state, items: [...state.items, action.payload] }; [cite: 21]
    case 'REMOVELIVRE':
      return { 
        ...state, 
        items: state.items.filter(isbn => isbn !== action.payload) [cite: 22]
      };
    default:
      return state; [cite: 21]
  }
};
export default panierReducer;
